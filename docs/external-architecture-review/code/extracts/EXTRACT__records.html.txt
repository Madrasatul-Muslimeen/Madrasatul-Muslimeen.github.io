   469      function entryRowHtml(chunkKey, entryKey, e) {
   470        const trackableName = trackableList.find((t) => t.id === e.trackableId);
   471        const domainNames = (e.domainIds ?? []).map((id) => domainList.find((d) => d.id === id)).filter(Boolean)
   472          .map((d) => `<span class="tag">${langText(d.name, getAppLang(), d.id)}</span>`).join("");
   473        const pillClass = `pill pill-${e.confirmState}`;
   474        // Round 11 -- "View as" wired up for real: a Student preview can't
   475        // confirm their own attainment (Architecture s6's role table), so
   476        // these actions just don't render while canConfirm is false.
   477        const actions = !canConfirm
   478          ? ""
   479          : e.confirmState === "pending"
   480            ? `<button class="small confirmBtn" data-chunk="${chunkKey}" data-entry="${entryKey}">${t("Confirm")}</button>
   481               <button class="small secondary returnBtn" data-chunk="${chunkKey}" data-entry="${entryKey}">${t("Return")}</button>`
   482            : e.confirmState === "returned"
   483              ? `<button class="small confirmBtn" data-chunk="${chunkKey}" data-entry="${entryKey}">${t("Confirm anyway")}</button>`
   484              : "";
   485        return `<tr>
   486          <td>${unitKeyLabel(entryKey.split("::")[0])}</td>
   487          <td>${trackableName ? langText(trackableName.name, getAppLang(), e.trackableId) : e.trackableId}</td>
   488          <td class="status-${e.claimedStatus}">${statusLabel(e.claimedStatus)}</td>
   489          <td class="status-${e.confirmedStatus}">${e.confirmedStatus ? statusLabel(e.confirmedStatus) : "—"}</td>
   490          <td><span class="${pillClass}">${confirmStateLabel(e.confirmState)}</span>${e.returnNote ? `<div style="font-size:0.78rem; color:#9c3b2e;">${e.returnNote}</div>` : ""}</td>
   491          <td>${domainNames || "—"}</td>
   492          <td>${e.notes || ""}</td>
   493          <td>${actions}</td>
   494        </tr>`;
